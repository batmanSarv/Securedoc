cmake_minimum_required(VERSION 3.14)
project(SecurDoc LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_FLAGS "${CMAKE_CXX_FLAGS} -Wall -Wextra -O2")
find_package(PkgConfig REQUIRED)
pkg_check_modules(SODIUM REQUIRED libsodium)
add_executable(securdoc src/securedoc.cpp src/header.cpp src/crypto.cpp)
target_link_libraries(securdoc PRIVATE ${SODIUM_LIBRARIES})
target_include_directories(securdoc PRIVATE include ${SODIUM_INCLUDE_DIRS})
